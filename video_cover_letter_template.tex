% Extract a YouTube video id from \VideoUrl.
% Supported forms: https://youtube.com/shorts/<id> and https://www.youtube.com/watch?v=<id>
\newcommand{\VideoIdTmp}{}
\newcommand{\VideoId}{}
\IfSubStr{\VideoUrl}{/shorts/}{\StrBehind{\VideoUrl}{/shorts/}[\VideoIdTmp]}{\StrBehind{\VideoUrl}{v=}[\VideoIdTmp]}
\IfSubStr{\VideoIdTmp}{?}{\StrBefore{\VideoIdTmp}{?}[\VideoId]}{\edef\VideoId{\VideoIdTmp}}

% Download thumbnail on build (requires -shell-escape).
\newcommand{\VideoThumbPath}{build/video-cover-letter-thumb.jpg}
\newcommand{\FetchVideoThumb}{%
  \immediate\write18{mkdir -p build && (curl -fsSL -o "\VideoThumbPath" "https://i.ytimg.com/vi/\VideoId/maxresdefault.jpg" || curl -fsSL -o "\VideoThumbPath" "https://i.ytimg.com/vi/\VideoId/hqdefault.jpg")}
}
